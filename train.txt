import copy
import time
import torch
import numpy as np
import torch.nn as nn
import matplotlib.pyplot as plt
from sklearn.metrics import roc_auc_score, average_precision_score, f1_score
import pandas as pd
import seaborn as sns
from scipy import stats
import torch.nn.functional as F


def train_model(model, optimizer, data_o, train_loader, val_loader, test_loader, args, use_curvature=True, squared=False, loss_curvature=True):
    m = torch.nn.Sigmoid() 
    loss_fct = torch.nn.BCELoss()
    loss_no_reduction = torch.nn.BCELoss(reduction='none')
    b_xent = nn.BCEWithLogitsLoss()
    loss_history = []
    max_auc = 0
    max_translation = 0
    alpha = 1.0
    lam = 0.5
    #loss_curvature = True

    if args.cuda:
        model.to('cuda')
        data_o.to('cuda')
        
    # Train model
    t_total = time.time()
    model_max = copy.deepcopy(model)
    print('Start Training...')
    #for translation in np.arange(-1.0,0.5,0.1):
    translation = 0.0
    for epoch in range(args.epochs):
        t = time.time()
        print('-------- Epoch ' + str(epoch + 1) + ' --------')
        y_pred_train = []
        y_label_train = []
        curv_list = []
        loss_list = []

        for i, (label, inp) in enumerate(train_loader):
            if args.cuda:
                label = label.cuda()

            model.train()
            optimizer.zero_grad()
            output = model(data_o,inp, use_curvature=use_curvature, squared=squared, translation=translation)
            pred = torch.squeeze(output)
            target = label.float().cuda()
            # compute standart BCE
            base_loss = F.binary_cross_entropy_with_logits(pred,target)
            # compute BCE per sample
            bce_per_sample = F.binary_cross_entropy_with_logits(pred, target, reduction = 'none')
            
            log = torch.sigmoid(pred)

            #calculate curvature for loss
            weights = torch.ones_like(bce_per_sample)
           # print("\n[DEBUG] loss_curvature =", loss_curvature)
            #print("[DEBUG] inp shape:", len(inp[0]))
            #print("[DEBUG] sample edge:", inp[0][0].item(), inp[1][0].item())
            #print("[DEBUG] edge_index_total shape:", data_o.edge_index_total.shape)

            if loss_curvature:
                for j in range(len(label)):
                    u, v = inp[0][j], inp[1][j]
                    mask_forward = (data_o.edge_index_total[0] == u) & (data_o.edge_index_total[1] == v)
                    mask_backward = (data_o.edge_index_total[0] == v) & (data_o.edge_index_total[1] == u)
                    combined_mask = mask_forward | mask_backward
                    if combined_mask.any():
                        edge_idx = combined_mask.nonzero(as_tuple=True)[0][0].item()
                        curvature = data_o.curva_total
                        curv = curvature[edge_idx]

                        #curv = (curv - curvature.mean()) / (curvature.std() + 1e-3)

                        weights[j] = 1 + alpha * torch.sigmoid(-curv)

            weights = weights / (weights.mean() + 1e-3)
            # compute curvature-weighted loss
            curv_loss = (weights * bce_per_sample).mean()
            #combine both losses
            loss_train = (1 - lam) * base_loss + lam * curv_loss
            
            loss_history.append(loss_train)
            loss_train.backward()
            optimizer.step()

            label_ids = label.to('cpu').numpy()
            y_label_train = y_label_train + label_ids.flatten().tolist()
            #y_pred_train = y_pred_train + output.flatten().tolist()
            y_pred_train = y_pred_train + log.flatten().tolist()

            if i % 100 == 0:
                print('epoch: ' + str(epoch + 1) + '/ iteration: ' + str(i + 1) + '/ loss_train: ' + str(
                    loss_train.cpu().detach().numpy()))

        roc_train = roc_auc_score(y_label_train, y_pred_train)

        # validation after each epoch
        if not args.fastmode:
            roc_val, prc_val, f1_val, loss_val, r_pos, df_val = test(model, val_loader, data_o, args, use_curvature=use_curvature, squared=squared, translation=translation, final_train=False)
            if roc_val > max_auc:
                model_max = copy.deepcopy(model)
                max_auc = roc_val
                

            print('epoch: {:04d}'.format(epoch + 1),
                      'loss_train: {:.4f}'.format(loss_train.item()),
                      'auroc_train: {:.4f}'.format(roc_train),
                      'loss_val: {:.4f}'.format(loss_val.item()),
                      'auroc_val: {:.4f}'.format(roc_val),
                      'auprc_val: {:.4f}'.format(prc_val),
                      'f1_val: {:.4f}'.format(f1_val),
                      'time: {:.4f}s'.format(time.time() - t))
        else:
            model_max = copy.deepcopy(model)

        if hasattr(torch.cuda, 'empty_cache'):
            torch.cuda.empty_cache()
                
    #plt.plot(loss_history)

    print("Optimization Finished!")
    print("Total time elapsed: {:.4f}s".format(time.time() - t_total))
    
    # Testing
    print('translation', max_translation)
    auroc_test, prc_test, f1_test, loss_test, r_pos, df_test = test(model_max, test_loader, data_o, args, use_curvature=use_curvature, squared=squared, translation=max_translation, final_train=True)
    print('loss_test: {:.4f}'.format(loss_test.item()), 'auroc_test: {:.4f}'.format(auroc_test),
          'auprc_test: {:.4f}'.format(prc_test), 'f1_test: {:.4f}'.format(f1_test))
    #torch.save(model_max.state_dict(), "best_model.pth")
    #with open(args.out_file, 'a') as f:
    #    f.write('{0}\t{1}\t{2}\t{7}\t{3:.4f}\t{4:.4f}\t{5:.4f}\t{6:.4f}\n'.format(
    #        args.in_file[5:8], args.seed, args.aggregator, loss_test.item(), auroc_test, prc_test, f1_test, args.feature_type))
    return df_test

def test(model, loader, data_o, args, use_curvature=True, squared=False, translation=0, final_train=False):
    model.eval()

    loss_fct = nn.BCEWithLogitsLoss()
    loss_no_reduction = nn.BCEWithLogitsLoss(reduction='none')
   
    y_pred = []
    y_label = []
    curv_list = []
    loss_list = []
    pred_list = [] #if 1 correlctly predicted edge
    edge_list = []

    
    with torch.no_grad():
        for i, (label, inp) in enumerate(loader):
            if args.cuda:
                label = label.cuda()
        
            output = model(data_o, inp,
               use_curvature=use_curvature,
               squared=squared,
               translation=translation)

            logits = torch.squeeze(output)

            loss = loss_fct(logits, label.float())
            loss_simple = loss_no_reduction(logits, label.float())

            probs = torch.sigmoid(logits)
            
            #get curvature
            if args.correlation and final_train:
                for j in range(len(label)):
                    u, v = inp[0][j], inp[1][j]
                    mask_forward = (data_o.edge_index_total[0] == u) & (data_o.edge_index_total[1] == v)
                    mask_backward = (data_o.edge_index_total[0] == v) & (data_o.edge_index_total[1] == u)
                    combined_mask = mask_forward | mask_backward
                    if combined_mask.any():
                        edge_idx = combined_mask.nonzero(as_tuple=True)[0][0].item()
                        edge_list.append(edge_idx)
                        curv_list.append(data_o.curva_total[edge_idx].item())
                        loss_list.append(loss_simple[j].item())
                        pred_list.append(log[j].item() >= 0.5)

            label_ids = label.to('cpu').numpy()
            y_label += label.detach().cpu().numpy().flatten().tolist()
            y_pred += probs.detach().cpu().numpy().flatten().tolist()
            outputs = np.asarray([1 if i else 0 for i in (np.asarray(y_pred) >= 0.5)])
            
         
            #print(len(y_pred))
        FN = np.sum((np.asarray(y_label) == 0) & (np.asarray(outputs) == 1)) #number of edges that were not predicted
        FP = np.sum((np.asarray(y_label) == 1) & (np.asarray(outputs) == 0)) #number of edges that were predicted incorrectly
        print('FN', FN, 'FP', FP)
        #curvature correlation
        (r, p_value, r_pos, p_value_pos, r_neg, p_value_neg) = (0,1,0,1,0,1)
        df = pd.DataFrame()
        if len(set(loss_list))>1 and final_train:
            df = pd.DataFrame({'loss': loss_list, 'curvature': curv_list, 'prediction': pred_list, 'edge_id' : edge_list})
            
            if args.correlation:
                r, p_value = stats.pearsonr(df['loss'], df['curvature'])
                r_spearman, p_value_spearman = stats.spearmanr(df['loss'], df['curvature'])
                print(f"Test correlation: {r: .4f}")
                print(f"p-value: {p_value: .10f}")
                print(f"Test spearman correlation: {r_spearman: .4f}")
                print(f"spearman p-value: {p_value_spearman: .10f}")
        
                mask_pos = df['curvature'] > 0
                if len(set(df['loss'][mask_pos]))>1:
                    r_pos, p_value_pos = stats.pearsonr(df['loss'][mask_pos], df['curvature'][mask_pos])
                    r_spearman_pos, p_value_spearman_pos = stats.spearmanr(df['loss'][mask_pos], df['curvature'][mask_pos])
                    print(f"Positive test correlation: {r_pos: .4f}")
                    print(f"Positive p-value: {p_value_pos: .10f}")
                    print(f"Test spearman correlation: {r_spearman_pos: .4f}")
                    print(f"spearman p-value: {p_value_spearman_pos: .10f}")
            
                mask_neg = df['curvature'] < 0
                if len(set(df['loss'][mask_neg]))>1:
                    r_neg, p_value_neg = stats.pearsonr(df['loss'][mask_neg], df['curvature'][mask_neg])
                    r_spearman_neg, p_value_spearman_neg = stats.spearmanr(df['loss'][mask_neg], df['curvature'][mask_neg])
                    print(f"Negative test correlation: {r_neg: .4f}")
                    print(f"Negative p-value: {p_value_neg: .10f}")
                    print(f"Test spearman correlation: {r_spearman_neg: .4f}")
                    print(f"spearman p-value: {p_value_spearman_neg: .10f}")
            
        corr = (r, p_value, r_pos, p_value_pos, r_neg, p_value_neg)
            
            
    return roc_auc_score(y_label, y_pred), average_precision_score(y_label, y_pred), f1_score(y_label, outputs), loss, corr, df
